# Calliope 0.7 lives in its own virtualenv (see Dockerfile) so its pinned
# dependency set doesn't collide with PyPSA / AdOpT in the system Python.
# The Go runner only ever calls the `calliope` CLI, so isolation is lossless.
#
# 0.7.0 is not yet on PyPI as a final release — only 0.7.0.dev* pre-releases —
# and the service emits Calliope 0.7 (model.yaml + additional_math.yaml) syntax,
# so we install the latest available 0.7 pre-release.
calliope==0.7.0.dev7

# calliope pins xarray<2024.4, but only says pandas>=2.1.3 — so the resolver
# grabs pandas 3.0, whose `to_xarray` demands xarray>=2024.10. That's an
# unsatisfiable combination. Cap pandas below 3.0 (2.2.x needs only an older
# xarray) so calliope can actually load model data.
pandas>=2.1.3,<2.3

# pyomo pin for calliope dev7. pyomo's HiGHS plugins cannot drive calliope dev7
# on ANY available pyomo (6.9.4/6.9.5/6.10 hit an add_block bug in
# contrib.solver's persistent interface; 6.8.x/appsi reject the call), so
# `solver: highs` runs through MEME's highspy bridge (internal/scripts/
# calliope_highs.py) instead. 6.8.2 is the version where cbc solves cleanly and
# general model building works; newer 6.9/6.10 regress it.
pyomo>=6.8.2,<6.9

# HiGHS inside the Calliope venv, driven directly by the highspy bridge for
# `solver: highs` (LP export -> highspy -> values back onto the pyomo model).
highspy
# HiPO, HiGHS's parallel interior-point solver (the bridge's default), loads
# its factorisation libraries (BLAS, METIS, AMD, RCM) from this package at
# run time; without it `solver: hipo` is rejected and the bridge uses IPX.
# Keep it on the same version as highspy.
highspy-extras
